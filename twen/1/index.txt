  <script>
    (() => {
      "use strict";

      const verticalVideos = document.querySelector("#verticalVideos");
      const horizontalVideos = document.querySelector("#horizontalVideos");
      const yearNode = document.querySelector("#year");
      const cooldownTimers = new Set();

      function getSmartlinkUrl() {
        return SMARTLINK.primary || SMARTLINK.secondary || "";
      }

      function getCooldownUntil() {
        const raw = localStorage.getItem(AD_CONFIG.storageKey);
        const value = Number(raw);
        return Number.isFinite(value) ? value : 0;
      }

      function setCooldown() {
        const until = Date.now() + AD_CONFIG.cooldownMs;
        localStorage.setItem(AD_CONFIG.storageKey, String(until));
        return until;
      }

      function isAdReady() {
        return Boolean(getSmartlinkUrl()) && Date.now() >= getCooldownUntil();
      }

      function updateAllAdLayers() {
        document.querySelectorAll(".player-ad-layer").forEach((layer) => {
          layer.hidden = !isAdReady();
        });
      }

      function scheduleGlobalRefresh(until) {
        const remaining = until - Date.now();

        if (remaining <= 0) {
          updateAllAdLayers();
          return;
        }

        const timer = window.setTimeout(() => {
          cooldownTimers.delete(timer);
          updateAllAdLayers();
        }, remaining + 20);

        cooldownTimers.add(timer);
      }

      function openSmartlink() {
        const url = getSmartlinkUrl();
        if (!url || !isAdReady()) return;

        // Lock the global cooldown before opening the Smartlink.
        const until = setCooldown();
        updateAllAdLayers();
        scheduleGlobalRefresh(until);

        window.open(url, "_blank", "noopener,noreferrer");
      }

      function captureFirstFrame(videoElement, applyThumbnail) {
        if (!videoElement.currentSrc || videoElement.dataset.thumbnailReady === "true") {
          return;
        }

        const capture = document.createElement("video");
        capture.muted = true;
        capture.playsInline = true;
        capture.preload = "auto";
        capture.src = videoElement.currentSrc;

        const cleanup = () => {
          capture.removeAttribute("src");
          capture.load();
        };

        capture.addEventListener("loadeddata", () => {
          try {
            capture.currentTime = 0;
          } catch {
            cleanup();
          }
        }, { once: true });

        capture.addEventListener("seeked", () => {
          try {
            const width = capture.videoWidth;
            const height = capture.videoHeight;
            if (!width || !height) {
              cleanup();
              return;
            }

            const canvas = document.createElement("canvas");
            canvas.width = width;
            canvas.height = height;

            const context = canvas.getContext("2d", { alpha: false });
            context.drawImage(capture, 0, 0, width, height);
            applyThumbnail(canvas.toDataURL("image/jpeg", 0.78));
            videoElement.dataset.thumbnailReady = "true";
          } catch {
            // A remote video without CORS permission may block canvas extraction.
            // In that case the browser's native first-frame preview remains available.
          } finally {
            cleanup();
          }
        }, { once: true });

        capture.addEventListener("error", cleanup, { once: true });
        capture.load();
      }

      function prepareThumbnail(videoElement) {
        if (!videoElement.currentSrc || videoElement.dataset.thumbnailStarted === "true") {
          return;
        }

        videoElement.dataset.thumbnailStarted = "true";

        const applyThumbnail = (poster) => {
          if (!videoElement.poster) {
            videoElement.poster = poster;
          }
        };

        videoElement.addEventListener("loadeddata", () => {
          captureFirstFrame(videoElement, applyThumbnail);
        }, { once: true });

        if (videoElement.readyState >= 2) {
          captureFirstFrame(videoElement, applyThumbnail);
        }
      }

      function createVideoCard(video) {
        const card = document.createElement("article");
        card.className = "video-card";

        const label = `Video ${String(video.id).padStart(2, "0")}`;
        const description = video.description || "Enjoy watching.";
        const source = video.videoUrl
          ? `<source src="${escapeHtml(video.videoUrl)}" type="video/mp4">`
          : "";
        const customPoster = video.thumbnail ? ` poster="${escapeHtml(video.thumbnail)}"` : "";

        card.innerHTML = `
          <div class="video-card__label">${label}</div>

          <div class="player-wrap">
            <video controls playsinline preload="metadata"${customPoster}>
              ${source}
            </video>

            <button
              class="player-ad-layer"
              type="button"
              aria-label="Advertisement"
              aria-hidden="false"
            >
              <span class="player-ad-badge">Advertisement</span>
            </button>
          </div>

          <div class="video-card__footer">
            <h2 class="video-card__title">${escapeHtml(video.title)}</h2>
            <p class="video-card__hint">${escapeHtml(description)}</p>
          </div>
        `;

        const adLayer = card.querySelector(".player-ad-layer");
        const videoElement = card.querySelector("video");

        adLayer.addEventListener("click", openSmartlink);

        videoElement.addEventListener("loadedmetadata", () => {
          adLayer.hidden = !isAdReady();
        });

        // Native HTML5 video does not autoplay. Thumbnail capture is attempted
        // from the video's first frame without starting playback.
        if (video.videoUrl && !video.thumbnail) {
          prepareThumbnail(videoElement);
        }

        return card;
      }

      function escapeHtml(value) {
        return String(value)
          .replaceAll("&", "&amp;")
          .replaceAll("<", "&lt;")
          .replaceAll(">", "&gt;")
          .replaceAll('"', "&quot;")
          .replaceAll("'", "&#039;");
      }

      function renderVideos() {
        if (!Array.isArray(VIDEOS)) return;

        VIDEOS.slice(0, 5).forEach((video) => {
          verticalVideos.appendChild(createVideoCard(video));
        });

        VIDEOS.slice(5, 10).forEach((video) => {
          horizontalVideos.appendChild(createVideoCard(video));
        });

        updateAllAdLayers();
        scheduleGlobalRefresh(getCooldownUntil());
      }

      window.addEventListener("storage", (event) => {
        if (event.key === AD_CONFIG.storageKey) {
          updateAllAdLayers();
          scheduleGlobalRefresh(getCooldownUntil());
        }
      });

      yearNode.textContent = new Date().getFullYear();
      renderVideos();
    })();
  </script>
